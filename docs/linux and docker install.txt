-- As root, update the system and set a password for the `root` user. This is necessary to ensure the system is up-to-date and secure.

apt update
apt upgrade
passwd
shutdown -r now

-- Install Docker and Docker Compose. This is necessary to run the Apensum stack.

sudo apt install ca-certificates curl
sudo install -m 0755 -d /etc/apt/keyrings
sudo curl -fsSL https://download.docker.com/linux/ubuntu/gpg -o /etc/apt/keyrings/docker.asc
sudo chmod a+r /etc/apt/keyrings/docker.asc
sudo apt install ca-certificates curl
sudo install -m 0755 -d /etc/apt/keyrings

# Add the repository to Apt sources:

sudo tee /etc/apt/sources.list.d/docker.sources <<EOF
Types: deb
URIs: https://download.docker.com/linux/ubuntu
Suites: $(. /etc/os-release && echo "${UBUNTU_CODENAME:-$VERSION_CODENAME}")
Components: stable
Architectures: $(dpkg --print-architecture)
Signed-By: /etc/apt/keyrings/docker.asc
EOF

sudo apt update
sudo apt install docker-ce docker-ce-cli containerd.io docker-buildx-plugin docker-compose-plugin
sudo apt autoremove
sudo docker run hello-world
adduser ubuntu
groupadd docker
sudo usermod -aG docker ubuntu
shutdown -r now

-- Add the `ubuntu` user to the sudoers file to allow it to run commands with elevated privileges. This is necessary for administrative tasks.

   apt install vim
   vi /etc/sudoers

-- Enable the firewall and allow only the necessary ports (SSH, HTTP, HTTPS). This is important for security.

   apt install ufw
   sudo ufw default deny incoming
   sudo ufw default allow outgoing
   sudo ufw allow 22/tcp      # SSH
   sudo ufw allow 80/tcp      # HTTP (Caddy, Let's Encrypt certificates)
   sudo ufw allow 443/tcp     # HTTPS
   sudo ufw enable            # asks for confirmation: answer y
   sudo ufw status verbose
   sudo apt install net-tools

-- As ubuntu user, generate an SSH key pair (private + public) and copy the public key to the server's `~/.ssh/authorized_keys` file. This allows passwordless SSH login.

mkdir .ssh
cd .ssh
vi id_ed25519.pub
mv id_ed25519.pub authorized_keys
docker ps
sudo ls -al
docker ps

-- Create directories for the various Apensum environments (production, demo, pre-production, test) under `/opt`. Set the owner and group to `ubuntu` to ensure proper permissions.

sudo install -d -o ubuntu -g ubuntu /opt/edge /opt/apensum-prod /opt/apensum-demo /opt/apensum-preprod /opt/apensum-test
